Log signal

curl -X POST http://localhost:5270/api/signals/logs \
  -H "Content-Type: application/json" \
  -d '{
    "source": "checkout-api",
    "level": "Error",
    "message": "HTTP 500 during checkout"
  }'

Metric signal

curl -X POST http://localhost:5270/api/signals/metrics \
  -H "Content-Type: application/json" \
  -d '{
    "metricName": "request_error_rate",
    "serviceName": "checkout-api",
    "environment": "production",
    "value": 42.6
  }'
